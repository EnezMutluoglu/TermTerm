cmake_minimum_required(VERSION 3.20)
project(termterm_windows_vpn_engines LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Use vpn-engine/CMakeLists.txt for the isolated Linux engines")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT OPENVPN_CORE OR NOT OPENCONNECT_INCLUDE_DIR)
  message(FATAL_ERROR "Pinned OPENVPN_CORE and audited OPENCONNECT_INCLUDE_DIR are required")
endif()
list(APPEND CMAKE_MODULE_PATH "${OPENVPN_CORE}/cmake")
include(findcoredeps)
find_package(jsoncpp CONFIG REQUIRED)
add_library(termterm-windows-openvpn-core STATIC "${OPENVPN_CORE}/client/ovpncli.cpp")
add_core_dependencies(termterm-windows-openvpn-core)
target_compile_definitions(termterm-windows-openvpn-core PRIVATE OPENVPN_EXTERNAL_TUN_FACTORY HAVE_JSON OPENVPN_CORE_API_VISIBILITY_HIDDEN)
target_link_libraries(termterm-windows-openvpn-core JsonCpp::JsonCpp)
add_executable(termterm-openvpn "../openvpn.cpp")
add_core_dependencies(termterm-openvpn)
target_link_libraries(termterm-openvpn termterm-windows-openvpn-core JsonCpp::JsonCpp)
add_executable(termterm-openconnect "../openconnect.cpp")
target_include_directories(termterm-openconnect PRIVATE "${OPENCONNECT_INCLUDE_DIR}")
target_compile_definitions(termterm-openconnect PRIVATE _WIN32_WINNT=0x0602)
target_link_libraries(termterm-openconnect JsonCpp::JsonCpp ws2_32)
add_executable(termterm-packet-selftest packet-selftest.cpp)
enable_testing()
add_test(NAME packet-validation COMMAND termterm-packet-selftest)
add_executable(termterm-sso-selftest sso-selftest.cpp)
target_link_libraries(termterm-sso-selftest JsonCpp::JsonCpp)
target_compile_options(termterm-sso-selftest PRIVATE /UNDEBUG)
add_test(NAME sso-validation COMMAND termterm-sso-selftest)
add_executable(termterm-dns-selftest dns-selftest.cpp)
add_core_dependencies(termterm-dns-selftest)
target_link_libraries(termterm-dns-selftest JsonCpp::JsonCpp)
target_compile_options(termterm-dns-selftest PRIVATE /UNDEBUG)
add_test(NAME dns-validation COMMAND termterm-dns-selftest)
