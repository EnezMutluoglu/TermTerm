cmake_minimum_required(VERSION 3.20)
project(termterm_vpn_engine LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJC)
  add_executable(termterm-vpn-macos macos-platform.m)
  target_compile_options(termterm-vpn-macos PRIVATE -fobjc-arc)
  target_link_libraries(termterm-vpn-macos PRIVATE "-framework Foundation" "-framework Security" "-framework SystemConfiguration" bsm)
endif()
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin)$")
  message(FATAL_ERROR "Unix helpers support Linux or macOS; Windows uses windows/CMakeLists.txt")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPENCONNECT REQUIRED IMPORTED_TARGET openconnect)
find_package(jsoncpp CONFIG REQUIRED)
add_executable(termterm-openconnect openconnect.cpp)
target_link_libraries(termterm-openconnect PRIVATE PkgConfig::OPENCONNECT JsonCpp::JsonCpp pthread)
if(OPENVPN_CORE)
  list(APPEND CMAKE_MODULE_PATH "${OPENVPN_CORE}/cmake")
  include(findcoredeps)
  add_library(termterm-openvpn-core STATIC "${OPENVPN_CORE}/client/ovpncli.cpp")
  add_core_dependencies(termterm-openvpn-core)
  target_compile_definitions(termterm-openvpn-core PRIVATE USE_TUN_BUILDER HAVE_JSON OPENVPN_CORE_API_VISIBILITY_HIDDEN)
  target_link_libraries(termterm-openvpn-core JsonCpp::JsonCpp)
  add_executable(termterm-openvpn openvpn.cpp)
  add_core_dependencies(termterm-openvpn)
  target_link_libraries(termterm-openvpn termterm-openvpn-core JsonCpp::JsonCpp)
endif()
